## 回文数

- 题目 ID：212226
- 状态：已通过
- 来源：本地原始文件
- 独立源码：`回文数.c`

```c
#include <stdio.h>

int main(void)
{
    int count[10];
    int odd_count = 0;
    int odd_digit = -1;
    int d;
    int k;
    int half_count[10];
    int half_len = 0;
    int first = -1;
    char left[55];
    int device = 0; /* 当前写到 left 数组的哪个位置 */

    for (d = 0; d <= 9; d++) {
        scanf("%d", &count[d]);

        if (count[d] % 2 != 0) {
            odd_count++;
            odd_digit = d;
        }
    }

    /* 回文数最多只能有一个数码出现奇数次。 */
    if (odd_count > 1) {
        printf("-1\n");
        return 0;
    }

    /* 每种数码放一半到左半边，剩下的由镜像产生。 */
    for (d = 0; d <= 9; d++) {
        half_count[d] = count[d] / 2;
        half_len += half_count[d];
    }

    if (half_len > 0) {
        /* 首位不能是 0，所以先选最小的非零数码。 */
        for (d = 1; d <= 9; d++) {
            if (half_count[d] > 0) {
                first = d;
                break;
            }
        }

        /* 左半边全是 0，无法组成没有前导 0 的数。 */
        if (first == -1) {
            printf("-1\n");
            return 0;
        }

        left[device] = (char)('0' + first);
        device++;
        half_count[first]--;

        /* 其余数码从小到大排列，使整个数最小。 */
        for (d = 0; d <= 9; d++) {
            for (k = 0; k < half_count[d]; k++) {
                left[device] = (char)('0' + d);
                device++;
            }
        }
    }

    if (device == 0) {
        /* 只剩一个数码时，它本身就是一位回文数。 */
        printf("%d\n", odd_digit);
        return 0;
    }

    /* 输出左半边和可能的中间数码。 */
    for (d = 0; d < device; d++) {
        putchar(left[d]);
    }
    if (odd_count == 1) {
        putchar((char)('0' + odd_digit));
    }

    /* 按相反顺序输出左半边，得到回文的右半边。 */
    for (d = device - 1; d >= 0; d--) {
        putchar(left[d]);
    }
    putchar('\n');

    return 0;
}
```

## 233

- 题目 ID：15756
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`233.c`

```c
#include <stdio.h>
#include <string.h>

int main(void)
{
    int T, t;

    scanf("%d", &T);

    for (t = 0; t < T; t++) {
        char s1[55], s2[55];

        /* a[i] 和 b[i] 保存数字的每一位 */
        int a[55] = {0};
        int b[55] = {0};

        /* result 保存乘积，每一位也倒着存放 */
        int result[110] = {0};

        int len1, len2;
        int i, j, len;

        scanf("%s%s", s1, s2);

        len1 = strlen(s1);
        len2 = strlen(s2);

        /*
         * 把字符串倒过来转换成数字。
         *
         * 例如输入 s1 = "233"
         * 原来：s1[0]='2', s1[1]='3', s1[2]='3'
         * 转换后：a[0]=3, a[1]=3, a[2]=2
         *
         * 这样 a[0] 就表示个位，a[1] 表示十位。
         */
        for (i = 0; i < len1; i++) {
            a[i] = s1[len1 - 1 - i] - '0';
        }

        for (i = 0; i < len2; i++) {
            b[i] = s2[len2 - 1 - i] - '0';
        }

        /*
         * 模拟竖式乘法。
         *
         * a[i] 和 b[j] 相乘后，要加到结果的 i+j 位。
         */
        for (i = 0; i < len1; i++) {
            for (j = 0; j < len2; j++) {
                result[i + j] += a[i] * b[j];
            }
        }

        /*
         * 处理进位。
         *
         * result[i] / 10 是进位；
         * result[i] % 10 是当前位保留下来的数字。
         */
        for (i = 0; i < 109; i++) {
            result[i + 1] += result[i] / 10;
            result[i] = result[i] % 10;
        }

        /*
         * 找到最高位。
         * 例如结果是 0054289，要把前面的 00 去掉。
         */
        len = len1 + len2;
        while (len > 0 && result[len] == 0) {
            len--;
        }

        /* 从最高位开始输出 */
        for (i = len; i >= 0; i--) {
            printf("%d", result[i]);
        }
        printf("\n");
    }

    return 0;
}
```

## 数位五五

- 题目 ID：22168
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`数位五五.c`

```c
#include <stdio.h>

int main()
{
    int a, b;
    scanf("%d %d", &a, &b);

    // 保证 a 比较小，b 比较大
    if (a > b) {
        int temp = a;
        a = b;
        b = temp;
    }

    int answer = 0;

    // 依次检查 a 到 b 的每个数
    for (int i = a; i <= b; i++) {
        int number = i;
        int sum = 0;

        // 取出每一位，并累加
        while (number > 0) {
            sum = sum + number % 10;
            number = number / 10;
        }

        // 判断各位数字之和是否为 5 的倍数
        if (sum % 5 == 0) {
            answer++;
        }
    }

    printf("%d\n", answer);
    return 0;
}
```

## 素数回文

- 题目 ID：14703
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`素数回文.c`

```c
#include <stdio.h>
#include <string.h>

typedef unsigned long long ull;

// 判断 n 是否为素数
int is_prime(ull n)
{
    if (n < 2)
        return 0;

    if (n == 2)
        return 1;

    // 大于 2 的偶数都不是素数
    if (n % 2 == 0)
        return 0;

    // 只检查奇数因子
    for (ull d = 3; d <= n / d; d += 2) {
        if (n % d == 0) {
            return 0;
        }
    }

    return 1;
}

int main(void)
{
    char s[20];

    scanf("%19s", s);

    int length = (int)strlen(s);
    ull palindrome = 0;

    // 第一步：把原来的数 t 放入 palindrome
    // 例如 "127" 变成 127
    for (int i = 0; i < length; i++) {
        palindrome = palindrome * 10 + (s[i] - '0');
    }

    // 第二步：从原数倒数第二位开始，依次加到末尾
    // 例如 "127" 接着加入 '2'、'1'，得到 12721
    for (int i = length - 2; i >= 0; i--) {
        palindrome = palindrome * 10 + (s[i] - '0');
    }

    if (is_prime(palindrome)) {
        printf("prime\n");
    } else {
        printf("noprime\n");
    }

    return 0;
}
```

## 日历中的数字

- 题目 ID：13584
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`日历中的数字.c`

```c
#include <stdio.h>

int main(void)
{
    int year, month, x;

    // 每次循环读取一组数据，读到文件结束为止
    while (scanf("%d %d %d", &year, &month, &x) == 3) {
        // days[1] 表示1月的天数，days[2] 表示2月的天数……
        int days[13] = {
            0,
            31, 28, 31, 30, 31, 30,
            31, 31, 30, 31, 30, 31
        };

        // 闰年时，2月有29天
        if ((year % 400 == 0) ||
            (year % 4 == 0 && year % 100 != 0)) {
            days[2] = 29;
        }

        int answer = 0;
        int day, i;
        char date[20];

        // 从1号一直数到这个月的最后一天
        for (day = 1; day <= days[month]; day++) {
            /*
             * 生成日期字符串：
             * %d  输出年份
             * %02d 输出至少两位，不足两位时前面补0
             */
            sprintf(date, "%d-%02d-%02d", year, month, day);

            // 检查日期中的每一个字符
            for (i = 0; date[i] != '\0'; i++) {
                if (date[i] - '0' == x) {
                    answer++;
                }
            }
        }

        printf("%d\n", answer);
    }

    return 0;
}
```

## 数字统计

- 题目 ID：16599
- 状态：未通过（原题记录20分）

```c
#include <stdio.h>

int main(void)
{
    int left, right;
    int answer = 0;

    scanf("%d %d", &left, &right);
    for (int value = left; value <= right; value++) {
        int current = value;
        while (current > 0) {
            if (current % 10 == 2) answer++;
            current /= 10;
        }
    }

    printf("%d\n", answer);
    return 0;
}
```
