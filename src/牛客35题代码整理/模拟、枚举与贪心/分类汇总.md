```c
#include <stdio.h>
#include <string.h>
#define MAX_LEN 100005
int main(){
    char s[MAX_LEN];
    char digits[5] = {'0', '2', '4', '6', '9'};
    char words[5][5] = {"zero", "two", "four", "six", "nine"};
    int word_len[5] = {4, 3, 4, 3, 4};
    int T;
    int test;
    scanf("%d", &T);
    for (test = 0; test < T; test++) {
        int i;
        scanf("%100004s", s);
        for (i = 0; s[i] != '\0'; i++) {
            int k;
            for (k = 0; k < 5; k++) {
                if (s[i] == digits[k] ||
                    strncmp(&s[i], words[k], word_len[k]) == 0) {
                    putchar(digits[k]);
                    break;
                }
            }
        }
        putchar('\n');
    }
    return 0;
}
```

## 安卓图案解锁

- 题目 ID：13585
- 状态：已通过
- 来源：本地原始文件
- 独立源码：`安卓图案解锁.c`

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int main(void) {
    char password[35];
    int used[10];
    while (scanf("%34s", password) == 1) {
        memset(used, 0, sizeof(used));
        int valid = 1;
        int length = (int)strlen(password);
        for (int i = 0; i < length && valid; ++i) {
            int current = password[i] - '0';
            if (current < 1 || current > 9 || used[current]) {
                valid = 0;
                break;
            }
            if (i > 0) {
                int previous = password[i - 1] - '0';
                int previous_row = (previous - 1) / 3;
                int previous_column = (previous - 1) % 3;
                int current_row = (current - 1) / 3;
                int current_column = (current - 1) % 3;
                int row_distance = abs(previous_row - current_row);
                int column_distance = abs(previous_column - current_column);

                if (row_distance % 2 == 0 && column_distance % 2 == 0) {
                    int middle = (previous + current) / 2;

                    if (!used[middle]) {
                        valid = 0;
                        break;
                    }
                }
            }

            used[current] = 1;
        }

        puts(valid ? "YES" : "NO");
    }

    return 0;
}
```

## 四舍五入

- 题目 ID：200369
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`四舍五入.c`

```c
#include <stdio.h>
#include <string.h>

#define MAXN 200005

char s[MAXN];

void addOne(int *length)
{
    int i = *length - 1;
    while (i >= 0 && s[i] == '9') {
        s[i] = '0';
        i--;
    }

    if (i >= 0) {
        s[i]++;
    } else {
        for (i = *length; i >= 0; i--) {
            s[i + 1] = s[i];
        }
        s[0] = '1';
        (*length)++;
    }
}

int main(void)
{
    int n;
    int length;
    long long t;

    scanf("%d%lld", &n, &t);
    scanf("%s", s);
    length = (int)strlen(s);

    char *dotPointer = strchr(s, '.');
    if (dotPointer == NULL) {
        puts(s);
        return 0;
    }

    int dot = (int)(dotPointer - s);
    int p = dot + 1;
    while (p < length && s[p] < '5') {
        p++;
    }
    if (p == length) p = -1;

    while (p != -1 && t > 0) {
        t--;

        if (p == dot + 1) {
            s[dot] = '\0';
            length = dot;
            addOne(&length);
            p = -1;
        } else {
            s[p] = '\0';
            length = p;

            int i = p - 1;
            while (i > dot && s[i] == '9') {
                s[i] = '0';
                i--;
            }

            if (i == dot) {
                s[dot] = '\0';
                length = dot;
                addOne(&length);
                p = -1;
            } else {
                s[i]++;
                p = s[i] >= '5' ? i : -1;
            }
        }
    }

    if (strchr(s, '.') != NULL) {
        length = (int)strlen(s);
        while (length > 0 && s[length - 1] == '0') length--;
        if (length > 0 && s[length - 1] == '.') length--;
        s[length] = '\0';
    }

    puts(s);
    return 0;
}
```

## 从后台研发到跑路

- 题目 ID：13588
- 状态：未通过（75分）
- 来源：本地原始文件
- 独立源码：`从后台研发到跑路.c`

```c
#include <stdio.h>
#define MAX_LEN 6000
int main(){
    char s[MAX_LEN];
    int n = 0;
    int ch;
    int i;
    int in_string = 0;
    while ((ch = getchar()) != EOF) {
        s[n] = (char)ch;
        n++;
    }
    i = 0;
    while (i < n) {
        if (in_string) {
            putchar(s[i]);
            if (s[i] == '"' && (i == 0 || s[i - 1] != '\\')) {
                in_string = 0;
            }
            i++;
        } else {
            if (s[i] == '"') {
                in_string = 1;
                putchar(s[i]);
                i++;
            } else if (i + 1 < n && s[i] == '/' && s[i + 1] == '*') {
                int j = i + 2;
                while (j + 1 < n && !(s[j] == '*' && s[j + 1] == '/')) {
                    j++;
                }
                if (j + 1 < n) {
                    i = j + 2;
                } else {
                    putchar(s[i]);
                    i++;
                }
            } else {
                putchar(s[i]);
                i++;
            }
        }
    }
    return 0;
}
```
