## 好串

- 题目 ID：21874
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`好串.c`

```c
#include <stdio.h>

int main(void) {
    char s[100005];

    scanf("%100000s", s);

    int unmatched_a = 0;  // 当前还未匹配的 a 的数量

    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == 'a') {
            unmatched_a++;
        } else if (s[i] == 'b') {
            if (unmatched_a == 0) {
                // 没有 a 可以与这个 b 配对
                printf("Bad\n");
                return 0;
            }
            unmatched_a--;
        } else {
            // 题目保证只有小写字母，但出现其他字符也不是好串
            printf("Bad\n");
            return 0;
        }
    }

    if (unmatched_a == 0) {
        printf("Good\n");
    } else {
        // 还有 a 没有与 b 配对
        printf("Bad\n");
    }

    return 0;
}
```

## 拼数

- 题目 ID：16783
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`拼数.c`

```c
#include <stdio.h>
#include <string.h>

#define MAX_N 20
#define MAX_LEN 100

int main(void)
{
    char nums[MAX_N][MAX_LEN + 1];
    int n;

    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        scanf("%s", nums[i]);
    }

    /* 冒泡排序 */
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            char first[2 * MAX_LEN + 1];
            char second[2 * MAX_LEN + 1];

            /* first = nums[j] + nums[j+1] */
            strcpy(first, nums[j]);
            strcat(first, nums[j + 1]);

            /* second = nums[j+1] + nums[j] */
            strcpy(second, nums[j + 1]);
            strcat(second, nums[j]);

            /* 如果前后拼接结果更小，就交换 */
            if (strcmp(first, second) < 0) {
                char temp[MAX_LEN + 1];

                strcpy(temp, nums[j]);
                strcpy(nums[j], nums[j + 1]);
                strcpy(nums[j + 1], temp);
            }
        }
    }

    for (int i = 0; i < n; i++) {
        printf("%s", nums[i]);
    }
    printf("\n");

    return 0;
}
```

## 牛牛的三角形

#include <stdio.h>

int main(void) {
    int n;
    long long a[105];

    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        scanf("%lld", &a[i]);
    }

    // 枚举三个下标互不相同的元素
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            for (int k = j + 1; k < n; k++) {
                long long x = a[i];
                long long y = a[j];
                long long z = a[k];

                // 任意两边之和都大于第三边
                if (x + y > z && x + z > y && y + z > x) {
                    printf("%lld %lld %lld\n", x, y, z);
                    return 0;
                }
            }
        }
    }

    printf("No solution\n");
    return 0;
}

## 装进肚子

- 题目 ID：14721
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`装进肚子.c`

```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_N 100005

typedef struct {
    long long morning;  // 早上吃的甜蜜值
    long long evening;  // 晚上吃的甜蜜值
    long long extra;    // 早上比晚上多吃到的甜蜜值
} Chocolate;

// 按照 extra 从大到小排序
int compare(const void *p1, const void *p2)
{
    Chocolate *x = (Chocolate *)p1;
    Chocolate *y = (Chocolate *)p2;

    if (x->extra < y->extra)
        return 1;
    if (x->extra > y->extra)
        return -1;

    return 0;
}

int main(void)
{
    int n, k;
    Chocolate chocolates[MAX_N];
    long long answer = 0;

    scanf("%d %d", &n, &k);

    // 读取每块巧克力早上的甜蜜值
    for (int i = 0; i < n; i++) {
        scanf("%lld", &chocolates[i].morning);
    }

    // 读取晚上的甜蜜值，并计算早上比晚上多出的甜蜜值
    for (int i = 0; i < n; i++) {
        scanf("%lld", &chocolates[i].evening);
        chocolates[i].extra =
            chocolates[i].morning - chocolates[i].evening;
    }

    // 按 extra 从大到小排序
    qsort(chocolates, n, sizeof(Chocolate), compare);

    // 前 k 个在早上吃，其余在晚上吃
    for (int i = 0; i < n; i++) {
        if (i < k) {
            answer += chocolates[i].morning;
        } else {
            answer += chocolates[i].evening;
        }
    }

    printf("%lld\n", answer);

    return 0;
}
```

        for (p2 = 1; p2 <= 8; p2++) {
