- 题目 ID：214218
- 状态：已通过
- 来源：依据题面补写
- 独立源码：`涂小天与他的画.c`

```c
#include <stdio.h>

int main(void)
{
    int testCase;
    scanf("%d", &testCase);

    while (testCase--) {
        int n;
        scanf("%d", &n);

        int middle = n / 2;
        for (int row = 0; row < n; row++) {
            int stars = row <= middle ? 2 * row + 1 : 2 * (n - 1 - row) + 1;
            int spaces = (n - stars) / 2;

            for (int i = 0; i < spaces; i++) putchar(' ');
            for (int i = 0; i < stars; i++) putchar('*');
            putchar('\n');
        }
    }

    return 0;
}
```

#include <stdio.h>

int main(void)
{
    char c;
    scanf(" %c", &c);

    int n = c - 'A' + 1;
    for (int row = 0; row < n; row++) {
        for (int column = 0; column < n - 1 - row; column++) {
            putchar(' ');
        }
        for (int column = 0; column <= row; column++) {
            putchar('A' + column);
        }
        for (int column = row - 1; column >= 0; column--) {
            putchar('A' + column);
        }
        putchar('\n');
    }

    return 0;
}

## 上下金字塔

- 题目 ID：22204
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`上下金字塔.c`

```c
#include <stdio.h>

void print_pyramid(int n)
{
    for (int row = 1; row <= n; row++) {
        for (int spaces = 1; spaces <= n - row; spaces++) {
            printf(" ");
        }
        for (int stars = 1; stars <= 2 * row - 1; stars++) {
            printf("*");
        }
        printf("\n");
    }

    for (int row = n - 1; row >= 1; row--) {
        for (int spaces = 1; spaces <= n - row; spaces++) {
            printf(" ");
        }
        for (int stars = 1; stars <= 2 * row - 1; stars++) {
            printf("*");
        }
        printf("\n");
    }
}

int main(void)
{
    int n1, n2;

    if (scanf("%d %d", &n1, &n2) != 2) {
        return 1;
    }

    print_pyramid(n1);
    putchar('\n');       // 两个图形之间空一行，可删除
    print_pyramid(n2);

    return 0;
}
```
