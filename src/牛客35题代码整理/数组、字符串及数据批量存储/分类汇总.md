## 比较月亮大小

- 题目 ID：15431
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`比较月亮大小.c`

```c
#include <stdio.h>

int main(void)
{
    int n;
    scanf("%d", &n);

    int previous = 0;
    int last = 0;
    for (int i = 0; i < n; i++) {
        previous = last;
        scanf("%d", &last);
    }

    if (n == 1) {
        if (last == 0) {
            puts("UP");
        } else if (last == 15) {
            puts("DOWN");
        } else {
            puts("-1");
        }
    } else if (last == 15) {
        puts("DOWN");
    } else if (last == 0) {
        puts("UP");
    } else if (previous < last) {
        puts("UP");
    } else {
        puts("DOWN");
    }

    return 0;
}
```

## 校门外的树

- 题目 ID：16649
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`校门外的树.c`

```c
#include <stdio.h>

int main(void)
{
    int L, M;
    int removed[10005] = {0};
    int l, r;
    int x;
    int answer = 0;

    scanf("%d%d", &L, &M);
    for (int i = 0; i < M; i++) {
        scanf("%d%d", &l, &r);
        for (x = l; x <= r; x++) {
            removed[x] = 1;
        }
    }

    for (x = 0; x <= L; x++) {
        if (removed[x] == 0) answer++;
    }

    printf("%d\n", answer);
    return 0;
}
```

## 约瑟夫环

- 题目 ID：22227
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`约瑟夫环.c`

```c
#include <stdio.h>

int main(void) {
    int n, k, m;

    if (scanf("%d %d %d", &n, &k, &m) != 3) {
        return 1;
    }

    /*
     * n 是取模的除数，必须大于 0。
     * 题目保证 n >= 2，这里额外检查以便程序更安全。
     */
    if (n <= 0 || m <= 0 || k < 0 || k >= n) {
        return 1;
    }

    int survivor = 0;

    /*
     * 循环从 i = 2 开始，且 i <= n，
     * 所以 i 永远大于 0，% i 不会除以 0。
     */
    for (int i = 2; i <= n; i++) {
        survivor = (survivor + m) % i;
    }

    /*
     * 前面已经保证 n > 0，
     * 因此这里的 % n 也是安全的。
     */
    int answer = (k + survivor) % n;

    printf("%d\n", answer);
    return 0;
}
```

## 牛牛的数学作业

- 题目 ID：205911
- 状态：已通过
- 来源：依据题面补写
- 独立源码：`牛牛的数学作业.c`

```c
#include <stdio.h>

int main(void)
{
    int testCase;
    scanf("%d", &testCase);

    while (testCase--) {
        int n;
        scanf("%d", &n);

        long long minimum = 0;
        long long maximum = 0;
        long long value = 0;
        long double average = 0.0L;
        long double squaredDifferenceSum = 0.0L;

        for (int i = 0; i < n; i++) {
            scanf("%lld", &value);
            if (i == 0 || value < minimum) minimum = value;
            if (i == 0 || value > maximum) maximum = value;

            long double difference = value - average;
            average += difference / (i + 1);
            squaredDifferenceSum += difference * (value - average);
        }

        long double variance = squaredDifferenceSum / n;
        printf("%lld %.3f\n", maximum - minimum, (double)variance);
    }

    return 0;
}
```

## 记数问题

- 题目 ID：16538
- 状态：已通过
- 来源：历史对话恢复
- 独立源码：`记数问题.c`

```c
#include <stdio.h>

int main(void) {
    int n, x;
    long long count = 0;

    scanf("%d %d", &n, &x);

    for (int i = 1; i <= n; i++) {
        int number = i;

        while (number > 0) {
            if (number % 10 == x) {
                count++;
            }
            number /= 10;
        }
    }

    printf("%lld\n", count);
    return 0;
}
```
